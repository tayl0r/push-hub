# push-hub Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a self-hosted, multi-tenant push service that owns the `user → device tokens` store and sends by user id, using the `@tsteil/push-service` engine for APNs/FCM delivery.

**Architecture:** Fastify + Drizzle/Postgres. Each request authenticates with a per-app API key → resolves a **tenant** (its slug + a dedicated `@tsteil/push-service` engine built from that app's APNs/FCM credentials). `POST /v1/tokens` upserts a device; `POST /v1/send` expands `to: userId[]` → all that tenant's devices for those users, calls the engine, deletes rows the engine reports `invalid`, and returns Expo-shaped tickets. The engine is unchanged and public; push-hub is a thin store + fan-out + auth layer.

**Tech Stack:** TypeScript (ESM), Node ≥22, Fastify, Drizzle ORM + `postgres`, `@tsteil/push-service` (git dep), vitest, Docker + GitHub Actions.

## Global Constraints

- **Tenant isolation:** every token row and every send is scoped to the tenant resolved from the API key. An app can never read or push another app's users. Uniqueness is `(appSlug, token)`, so a registration can never evict another tenant's row — isolation is structural, not probabilistic.
- **API keys are compared by hash** (SHA-256 hex); plaintext keys never touch config or the DB.
- **Provider is derived** from `platform` (`ios → apns`, `android → fcm`) at send time — no `provider` column, no client-supplied field.
- **Dead-token cleanup is server-side and automatic:** delete exactly the rows the engine reports `invalid` (by `ref = push_tokens.id`); never delete on `failed`.
- **Fail-loud config:** a malformed present-provider block throws when its engine is built at startup → the deploy fails, not the first send.
- **ESM only** (`"type": "module"`), `strict` TypeScript, Node ≥22.
- **No secrets in the repo.** All tenant credentials + key hashes come from the `PUSH_HUB_TENANTS` env var. The repo is public; `.env.example` carries only the shape.

```ts
import { createPushService, type ApnsConfig, type FcmConfig, type PushService } from '@tsteil/push-service'

export interface Tenant {
  slug: string
  apiKeyHash: string
  engine: PushService
}

interface TenantInput {
  slug: string
  apiKeyHash: string
  apns?: ApnsConfig
  fcm?: FcmConfig
}

export function loadTenants(json: string): Map<string, Tenant> {
  const parsed = JSON.parse(json) as TenantInput[]
  const byHash = new Map<string, Tenant>()
  for (const t of parsed) {
    if (!t.slug || !t.apiKeyHash) throw new Error(`push-hub: tenant needs slug and apiKeyHash (got ${JSON.stringify(t)})`)
    if (!t.apns && !t.fcm) throw new Error(`push-hub: tenant ${t.slug} must configure at least one of apns/fcm`)
    // createPushService throws on a malformed present provider block — fail the deploy, not the first send.
    const engine = createPushService({ apns: t.apns, fcm: t.fcm })
    byHash.set(t.apiKeyHash, { slug: t.slug, apiKeyHash: t.apiKeyHash, engine })
  }
  return byHash
}
```

```ts
import { createHash } from 'node:crypto'
import type { FastifyReply, FastifyRequest, preHandlerHookHandler } from 'fastify'
import type { Tenant } from './config.js'

declare module 'fastify' {
  interface FastifyRequest { tenant: Tenant }
}

export function hashKey(key: string): string {
  return createHash('sha256').update(key).digest('hex')
}

export function requireTenant(tenants: Map<string, Tenant>): preHandlerHookHandler {
  return async (req: FastifyRequest, reply: FastifyReply) => {
    const header = req.headers.authorization
    // RFC 7235 auth schemes are case-insensitive; match "Bearer " loosely
    const key = header && /^bearer /i.test(header) ? header.slice(7).trim() : null
    if (!key) return reply.code(401).send({ error: 'missing bearer token' })
    const tenant = tenants.get(hashKey(key))
    if (!tenant) return reply.code(401).send({ error: 'invalid token' })
    req.tenant = tenant
  }
}
```

```ts
import { and, eq, inArray } from 'drizzle-orm'
import { pushTokens } from './db/schema.js'

type Db = any // drizzle instance; kept loose so store fns work with the app db and the test db

export async function registerToken(db: Db, appSlug: string, userId: string, token: string, platform: 'ios' | 'android'): Promise<void> {
  await db.insert(pushTokens)
    .values({ appSlug, userId, token, platform, lastSeenAt: new Date() })
    // conflict target is the (appSlug, token) unique constraint: a re-register within the same app reassigns the user;
    // the same token string under another app is a separate row (structural tenant isolation, never a cross-tenant evict)
    .onConflictDoUpdate({ target: [pushTokens.appSlug, pushTokens.token], set: { userId, platform, lastSeenAt: new Date() } })
}

export async function removeToken(db: Db, appSlug: string, token: string): Promise<number> {
  const deleted = await db.delete(pushTokens)
    .where(and(eq(pushTokens.appSlug, appSlug), eq(pushTokens.token, token)))
    .returning()
  return deleted.length
}

export async function tokensForUsers(db: Db, appSlug: string, userIds: string[]): Promise<Array<{ id: string; token: string; platform: 'ios' | 'android'; userId: string }>> {
  if (userIds.length === 0) return []
  return db.select({ id: pushTokens.id, token: pushTokens.token, platform: pushTokens.platform, userId: pushTokens.userId })
    .from(pushTokens)
    .where(and(eq(pushTokens.appSlug, appSlug), inArray(pushTokens.userId, userIds)))
}

export async function deleteByIds(db: Db, ids: string[]): Promise<void> {
  if (ids.length) await db.delete(pushTokens).where(inArray(pushTokens.id, ids))
}
```

```ts
    const parsed = registerBody.safeParse(req.body)
    if (!parsed.success) return reply.code(400).send({ error: 'invalid body', details: parsed.error.flatten() })
    const { userId, token, platform } = parsed.data
    await registerToken(deps.db, req.tenant.slug, userId, token, platform)
    return reply.send({ ok: true })
  })

    const parsed = removeBody.safeParse(req.body)
    if (!parsed.success) return reply.code(400).send({ error: 'invalid body' })
    const deleted = await removeToken(deps.db, req.tenant.slug, parsed.data.token)
    return reply.send({ ok: true, deleted })
  })
}
```

```ts
import type { FastifyInstance } from 'fastify'
import { z } from 'zod'
import * as Sentry from '@sentry/node'
import { requireTenant } from '../auth.js'
import type { Tenant } from '../config.js'
import { tokensForUsers, deleteByIds } from '../store.js'

const sendBody = z.object({
  to: z.union([z.string().min(1), z.array(z.string().min(1)).min(1)]),
  title: z.string().optional(),
  body: z.string().optional(),
  data: z.record(z.string()).optional(),
  badge: z.number().int().optional(),
  sound: z.string().optional(),
  priority: z.enum(['high', 'normal']).optional(),
})

export function sendRoutes(app: FastifyInstance, deps: { db: any; tenants: Map<string, Tenant> }) {
  app.post('/v1/send', { preHandler: requireTenant(deps.tenants) }, async (req, reply) => {
    const parsed = sendBody.safeParse(req.body)
    if (!parsed.success) return reply.code(400).send({ error: 'invalid body', details: parsed.error.flatten() })
    const { to, title, body, data, badge, sound, priority } = parsed.data
    const userIds = Array.isArray(to) ? to : [to]

    const rows = await tokensForUsers(deps.db, req.tenant.slug, userIds)
    if (rows.length === 0) return reply.send({ data: [] })

    const targets = rows.map(r => ({
      token: r.token,
      provider: r.platform === 'ios' ? ('apns' as const) : ('fcm' as const),
      ref: r.id,
    }))

    let results
    try {
      results = await req.tenant.engine.send({ tokens: targets, notification: { title, body, data, badge, sound, priority } })
    } catch (err) {
      req.log.error({ err }, 'push-hub: engine send failed (tenant misconfig?)')
      Sentry.captureException(err, { tags: { tenant: req.tenant.slug } })
      return reply.code(500).send({ error: 'send failed' })
    }

    const rowById = new Map(rows.map(r => [r.id, r]))
    await deleteByIds(deps.db, results.filter(r => r.status === 'invalid' && r.ref).map(r => r.ref!))

    const tickets = results.map(r => ({
      userId: r.ref ? rowById.get(r.ref)?.userId : undefined,
      status: r.status === 'delivered' ? 'ok' : 'error',
      ...(r.status === 'delivered' ? {} : { error: r.error ?? r.status }),
    }))
    return reply.send({ data: tickets })
  })
}
```
